% 논문 출력 모드 설정
%
% 이 값은 수동 컴파일할 때의 기본 모드입니다. 빌드 스크립트
% JKASE_main_build.cmd / JKASE_main_build.sh를 사용할 경우에는
% 스크립트가 \ANONYMIZED, \REVIEWCOPY, \FINALMODE 같은 명령을 주입하여
% 아래 기본값을 덮어쓸 수 있습니다.
%
%   submission : 투고본. 1단 편집, 저자명과 소속을 숨김.
%   review     : 심사본. 1단 편집, 저자명과 소속을 숨기고 심사용 표시 가능.
%   final      : 최종본. 2단 편집, 저자명과 소속을 모두 표시.
%
% 수동으로 최종본을 만들고 싶으면 아래 submission 줄을 주석 처리하고
% \def\PAPERMODE{final} 줄의 주석을 해제하면 됩니다.
% ==================================================================
\def\PAPERMODE{submission}
% \def\PAPERMODE{final}


% jkase.cls 안에는 \ifanonymizedsubmission, \ifreviewcopy, \iffinalcopy 같은
% 모드 판별용 boolean이 선언되어 있습니다. 여기서는 위에서 정한
% \PAPERMODE 문자열을 읽어 해당 boolean을 켭니다.
%
% \makeatletter ... \makeatother 구간에서는 @가 들어간 내부 명령
% 이름(\@jkpm 등)을 사용할 수 있습니다. 이 내부 임시 변수들은
% "현재 모드"와 "각 모드 이름"을 비교하기 위해서만 사용합니다.
\makeatletter
\edef\@jkpm{\PAPERMODE}
\edef\@jkpmS{submission}
\edef\@jkpmR{review}
\edef\@jkpmF{final}
\ifx\@jkpm\@jkpmS
  % 투고본은 익명 처리만 켭니다.
  \anonymizedsubmissiontrue
\fi
\ifx\@jkpm\@jkpmR
  % 심사본은 익명 처리와 심사용 사본 표시를 함께 켭니다.
  \anonymizedsubmissiontrue
  \reviewcopytrue
\fi
\ifx\@jkpm\@jkpmF
  % 최종본은 익명 처리를 끄고 최종본 플래그를 켭니다.
  \anonymizedsubmissionfalse
  \reviewcopyfalse
  \finalcopytrue
\fi
\makeatother

% 명령행 빌드 스크립트에서 모드를 지정하는 경우의 우선 처리입니다.
% 예를 들어 build.cmd final처럼 실행하면 스크립트가 \FINALMODE를 주입하고,
% 이 구간에서 최종본 설정이 강제로 적용됩니다. 즉, 파일 안의 기본
% \PAPERMODE 값보다 빌드 스크립트 인자가 우선합니다.
\ifdefined\ANONYMIZED
  \anonymizedsubmissiontrue
  \finalcopyfalse
\fi
\ifdefined\REVIEWCOPY
  \anonymizedsubmissiontrue
  \reviewcopytrue
  \finalcopyfalse
\fi
\ifdefined\FINALMODE
  \anonymizedsubmissionfalse
  \reviewcopyfalse
  \finalcopytrue
\fi

% 위에서 확정한 클래스 모드 boolean을 편의 boolean(\ifDocFinal 등)으로
% 옮겨 둡니다. 아래 표/그림 폭 조정 환경과 sub/ 본문 파일이 이 boolean을
% 사용합니다.
%   final      -> \ifDocFinal
%   review     -> \ifDocReview
%   그 외      -> \ifDocSubmission
\newif\ifDocFinal
\newif\ifDocReview
\newif\ifDocSubmission
\iffinalcopy
  \DocFinaltrue
\else
  \ifreviewcopy
    \DocReviewtrue
  \else
    \DocSubmissiontrue
  \fi
\fi

% ==================================================================
% 표/그림 폭 조정 도우미 환경
%
% 한국과학교육학회지 최종본은 2단 편집이므로, 표와 그림은 처음부터
% 2단 폭(1.0\textwidth) 또는 1단 폭(0.5\textwidth)을 정해 두는 것이
% 좋습니다. 아래 환경은 final 모드에서는 두 단을 가로지르는 표*/그림*
% (table*/figure*)로, submission/review 모드에서는 일반 표/그림으로
% 자동 전환합니다.
% ==================================================================
\newenvironment{ManuscriptWideTable}[1][htbp]{%
  \ifDocFinal\begin{table*}[#1]\else\begin{table}[#1]\fi
}{%
  \ifDocFinal\end{table*}\else\end{table}\fi
}

\newenvironment{ManuscriptWideFigure}[1][htbp]{%
  \ifDocFinal\begin{figure*}[#1]\else\begin{figure}[#1]\fi
}{%
  \ifDocFinal\end{figure*}\else\end{figure}\fi
}

\newlength{\ManuscriptHalfWidthLen}
\newcommand{\ManuscriptSetHalfWidth}{%
  \ifDocFinal
    \setlength{\ManuscriptHalfWidthLen}{\columnwidth}%
  \else
    \setlength{\ManuscriptHalfWidthLen}{0.5\textwidth}%
  \fi
}
\newcommand{\ManuscriptHalfWidth}{\ManuscriptHalfWidthLen}
\newenvironment{ManuscriptColumnTable}[1][htbp]{%
  \begin{table}[#1]
  \ManuscriptSetHalfWidth
  \noindent\begin{minipage}{\ManuscriptHalfWidth}
  \captionsetup{width=\linewidth,justification=raggedright,singlelinecheck=false}%
}{%
  \end{minipage}
  \end{table}
}
\newenvironment{ManuscriptColumnFigure}[1][htbp]{%
  \begin{figure}[#1]
  \ManuscriptSetHalfWidth
  \noindent\begin{minipage}{\ManuscriptHalfWidth}
  \captionsetup{width=\linewidth,justification=raggedright,singlelinecheck=false}%
}{%
  \end{minipage}
  \end{figure}
}

% 글꼴과 기본 문서 설정 대부분은 sub/0-preamble.tex에서 처리합니다.

% 보조 설정만 남겨 두었습니다.

% 모든 하위 파일(sub/*.tex)의 table 환경에 공통 적용되는 크기 조정입니다.
% 표가 본문보다 지나치게 커지지 않도록 table 시작 시 \small로 낮추고,

% 캡션에 쓰는 9pt 크기 글꼴(captionnine)을 정의합니다.
% 표/그림 캡션을 본문보다 약간 작게, 고딕(sf) 계열로 출력하기 위한 설정입니다.
\DeclareCaptionFont{captionnine}{\fontsize{9.0pt}{10.8pt}\selectfont}

% 표와 그림 캡션의 영문 라벨 형식입니다.
% name=Table/Figure     : "표", "그림" 대신 "Table", "Figure"로 출력.
% labelsep=period       : "Table 1."처럼 번호 뒤에 마침표 사용.
% justification=justified: 여러 줄 캡션을 양쪽 정렬.
% singlelinecheck=false : 한 줄 캡션도 동일한 정렬 규칙 적용.
\captionsetup[table]{name=Table,font={sf,captionnine},labelfont={sf,bf,captionnine},labelsep=period,justification=justified,singlelinecheck=false}
\captionsetup[figure]{name=Figure,font={sf,captionnine},labelfont={sf,bf,captionnine},labelsep=period,justification=justified,singlelinecheck=false}

% 캡션 폭을 상황별로 빠르게 바꾸기 위한 보조 명령입니다.
% 전체 폭 표/그림에는 \FullCaptionWidth, 2단 편집의 반폭 그림에는
% \HalfCaptionWidth, minipage나 resizebox 내부처럼 현재 줄폭 기준이
% 필요한 곳에는 \LocalCaptionWidth를 사용합니다.
\newcommand{\FullCaptionWidth}{\captionsetup{width=\textwidth,justification=justified,singlelinecheck=false}}
\newcommand{\HalfCaptionWidth}{\captionsetup{width=0.5\textwidth,justification=justified,singlelinecheck=false}}
\newcommand{\LocalCaptionWidth}{\captionsetup{width=\linewidth,justification=justified,singlelinecheck=false}}

% 표 아래 주석을 붙일 때 사용하는 명령입니다.
% \FullTableNote는 본문 전체 폭 기준, \LocalTableNote는 현재 환경의
% \linewidth 기준으로 주석 박스를 만듭니다.
\newcommand{\FullTableNote}[1]{%
  \par\vspace{0.3em}\noindent
  \begin{minipage}{\textwidth}
  \rmfamily\footnotesize #1\par
  \end{minipage}%
}
\newcommand{\LocalTableNote}[1]{%
  \par\vspace{0.3em}\noindent
  \begin{minipage}{\linewidth}
  \rmfamily\footnotesize #1\par
  \end{minipage}%
}
