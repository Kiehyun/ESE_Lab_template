% ---------- Name Spacing Commands ----------
% 아래 Lua 코드는 한글 이름의 글자 사이를 자동으로 벌려 주는 보조 기능입니다.
% 논문 정보만 바꾸는 사용자는 이 코드 블록을 수정하지 않아도 됩니다.
% 컴파일은 LuaLaTeX를 기준으로 합니다.
% The Lua code below automatically inserts spacing between Korean name characters.
% Users who only edit thesis information do not need to modify this block.
% This template is intended to be compiled with LuaLaTeX.

% 한글 이름 자간 공백 삽입 명령
% \SpacedName{\committeeChairKor}처럼 사용하면 각 글자 사이에 공백이 들어갑니다.
% Korean name spacing command
% Use it like \SpacedName{\committeeChairKor} to insert spaces between each character.
\directlua{
  function spacedKorName(s)
    local t = {}
    for _, c in utf8.codes(s) do
      table.insert(t, utf8.char(c))
    end
    return table.concat(t, "\\ ")
  end
}
\newcommand{\SpacedName}[1]{%
  \edef\@spn@tmp{#1}%
  \directlua{tex.sprint(spacedKorName("\luaescapestring{\@spn@tmp}"))}%
}

% 글자 사이에 정확히 반각(0.5em)을 넣는 명령. 예: 표지 전공 "지구과학교육전공".
% Insert exactly a half-em (0.5em) between characters. Used e.g. for the cover major line.
\directlua{
  function spacedHalfEm(s)
    local t = {}
    for _, c in utf8.codes(s) do
      table.insert(t, utf8.char(c))
    end
    return table.concat(t, "\\hspace{.5em}")
  end
}
\newcommand{\SpacedHalf}[1]{%
  \edef\@sph@tmp{#1}%
  \directlua{tex.sprint(spacedHalfEm("\luaescapestring{\@sph@tmp}"))}%
}

% 심사위원 이름은 3쪽 인준면에서 이름칸 폭에 맞춰 균등 배치합니다.
% 두 글자, 세 글자, 네 글자 이름이 섞여 있어도 같은 이름칸 안에 들어가도록
% 글자 사이에 늘어나는 공백(\hfill)을 넣습니다.
% 이름칸 자체가 너무 넓거나 좁으면 coverpage-thesis.tex의 \CommitteeNameWidth를 조정합니다.
% Committee names are evenly spaced in the approval page name column.
% Stretchable spaces (\hfill) keep two-, three-, and four-character Korean names aligned within the same name width.
% If the name column is too wide or narrow, adjust \CommitteeNameWidth in coverpage-thesis.tex.
\directlua{
	function spacedKorCommitteeName(s)
		local t = {}
		local count = 0
		for _, c in utf8.codes(s) do
			table.insert(t, utf8.char(c))
			count = count + 1
		end
		if count <= 1 then
			return table.concat(t)
		end
		return table.concat(t, "\\hfill{}")
	end
}
\newcommand{\SpacedCommitteeName}[1]{%
	\edef\@spn@committee@tmp{#1}%
	\directlua{tex.sprint(spacedKorCommitteeName("\luaescapestring{\@spn@committee@tmp}"))}%
}

% - 영어 문헌은 기존 \textcite, \parencite를 사용합니다.
% - 아래 설정은 \ktextcite와 \kparencite 안에서만 임시로 적용됩니다.

% \kparencite(괄호형) 저자 구분자: 영문과 동일하게 "&"  -> (홍길동 & 임꺽정, 2016)
